class PerformanceMetrics:
    def __init__(self):

        self.positions = {}  # Store positions with their corresponding trade IDs

    def update_trades(self, trade_id, position_id, profit_loss):
        if trade_id not in self.trades:
            self.trades[trade_id] = []
        self.trades[trade_id].append(profit_loss)

    def update_positions(self, position_id, trade_id):
        if position_id not in self.positions:
            self.positions[position_id] = []
        self.positions[position_id].append(trade_id)

    def calculate_pnl(self):
